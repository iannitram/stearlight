from pathlib import Path
import unittest

ROOT = Path(__file__).resolve().parents[1]
RUNNERS = tuple(
    line for line in (ROOT / 'build_files/quest/check.sh').read_text().splitlines()
    if line.strip() and not line.lstrip().startswith('#')
)
# Test files that check.sh must run. Later wiring drafts append entries.
REQUIRED = (
    'packages/quest-runtime/tests/desktop-manager.test.py',
    'packages/quest-runtime/tests/steam-channel.test.py',
    'packages/quest-runtime/tests/steam-channels-state.test.py',
    'packages/quest-runtime/tests/steam-client-update.test.py',
)


class CheckWiringTest(unittest.TestCase):
    def test_each_required_test_is_run_by_check(self):
        for path in REQUIRED:
            with self.subTest(path=path):
                self.assertTrue(any(path in line for line in RUNNERS),
                                path + ' is not run by check.sh')


if __name__ == '__main__':
    unittest.main()
