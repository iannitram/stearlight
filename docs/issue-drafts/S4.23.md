What: I want the Celeste renderer test to run the upgrade, restore and unknown-update paths with real SHA-256 digests. Today it patches `hashlib.sha256` globally so a tiny fixture hashes to the pinned game checksum, which means the comparison in the launcher never sees real bytes. Found in a code review. The fix is to let the caller pass the expected original digest, with the pinned value as the default, so the test passes a digest it computed itself.
Where: packages/quest-runtime/linux/fex-game-launcher.py:171-175 (prepare_celeste_memory; add an `original` parameter that defaults to the pinned digest); packages/quest-runtime/tests/celeste-memory.test.py:43-58 (the patched test). Code map: docs/code-map.md#quest-linux-session, docs/code-map.md#tests
Size: Easy (about 25 changed lines; 2 files; Python; Linux build host)
Blocked by: S4.19
Serves: none (cleanup)
Done when: the test no longer patches hashlib; it passes `original=<sha256 of its fixture>`; the production call in main() is unchanged and still uses the pinned digest; the test is red on the current code (the keyword does not exist) and green after the change.

## Tests (locked)
Path: packages/quest-runtime/tests/celeste-memory.test.py
```python
import importlib.util
import hashlib
import json
from pathlib import Path
import tempfile
import unittest

spec = importlib.util.spec_from_file_location("fex_launcher", Path(__file__).resolve().parents[1] / "linux/fex-game-launcher.py")
launcher = importlib.util.module_from_spec(spec)
spec.loader.exec_module(launcher)

ORIGINAL = b"original fixture"
ORIGINAL_SHA256 = hashlib.sha256(ORIGINAL).hexdigest()


def snapshot(folder):
    return {path.relative_to(folder).as_posix(): path.read_bytes()
            for path in sorted(folder.rglob("*")) if path.is_file()}


class CelesteMemoryTest(unittest.TestCase):
    def test_updated_game_without_fna3d_is_left_untouched(self):
        with tempfile.TemporaryDirectory() as folder:
            game = Path(folder)
            files = {"lib64/libSDL2.so": b"steam sdl", "Celeste.exe": b"steam exe", "Content/level.bin": b"level"}
            for name, data in files.items():
                (game / name).parent.mkdir(parents=True, exist_ok=True)
                (game / name).write_bytes(data)
            before = snapshot(game)
            self.assertEqual(len(before), 3)
            launcher.prepare_celeste_memory(game)
            launcher.prepare_celeste_memory(game, False)
            self.assertEqual(snapshot(game), before)

    def test_unknown_game_update_is_preserved(self):
        with tempfile.TemporaryDirectory() as folder:
            game = Path(folder)
            library = game / "lib64/libFNA3D.so.0"
            library.parent.mkdir()
            library.write_bytes(b"unrecognized Steam game update")
            launcher.prepare_celeste_memory(game)
            launcher.prepare_celeste_memory(game, False)
            self.assertEqual(library.read_bytes(), b"unrecognized Steam game update")
            self.assertFalse(library.with_name(library.name + ".quest-original").exists())

    def test_managed_renderer_upgrade_restore_and_unknown_game_update(self):
        with tempfile.TemporaryDirectory() as folder:
            game = Path(folder)
            library = game / "lib64/libFNA3D.so.0"
            library.parent.mkdir()
            library.write_bytes(ORIGINAL)
            library.chmod(0o755)
            runtime = game / "runtime"
            runtime.mkdir()

            def bundle(data):
                (runtime / "libFNA3D.so.0").write_bytes(data)
                (runtime / "profile.json").write_text(json.dumps({"sha256": hashlib.sha256(data).hexdigest()}))

            def prepare(enabled=True):
                launcher.prepare_celeste_memory(game, enabled, runtime, original=ORIGINAL_SHA256)

            bundle(b"renderer one")
            prepare()
            self.assertEqual(library.read_bytes(), b"renderer one")
            self.assertEqual(library.stat().st_mode & 0o777, 0o755)
            prepare()
            bundle(b"renderer two")
            prepare()
            self.assertEqual(library.read_bytes(), b"renderer two")
            prepare(False)
            self.assertEqual(library.read_bytes(), ORIGINAL)
            prepare()
            library.write_bytes(b"Steam update")
            prepare()
            prepare(False)
            self.assertEqual(library.read_bytes(), b"Steam update")

    def test_corrupt_bundle_does_not_touch_game(self):
        with tempfile.TemporaryDirectory() as folder:
            game = Path(folder)
            library = game / "lib64/libFNA3D.so.0"
            library.parent.mkdir()
            library.write_bytes(b"original")
            runtime = game / "runtime"
            runtime.mkdir()
            (runtime / "libFNA3D.so.0").write_bytes(b"corrupt")
            (runtime / "profile.json").write_text('{"sha256":"wrong"}')
            with self.assertRaisesRegex(ValueError, "checksum"):
                launcher.prepare_celeste_memory(game, runtime=runtime)
            self.assertEqual(library.read_bytes(), b"original")


if __name__ == "__main__":
    unittest.main()
```
