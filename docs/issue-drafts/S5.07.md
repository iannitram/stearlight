## Qwen brief
Expectation: Add `resolve_image_path(root, path)` to build_files/stage-frame-desktop.py. It follows symlinks one link at a time, as the current loop does, and returns the final path.
- An absolute link target is re-rooted under `root`, as now (`root / str(link).lstrip("/")`).
- A relative link is resolved against the directory of the link, as now.
- Before each step, take `os.path.normpath()` of the candidate. If the result is neither `root` nor below `root`, raise `ValueError(f'Symlink escapes the Frame image: {path}')`. The message must contain the word `escapes`.
- A path that is not a symlink returns itself.
`copy()` calls `resolve_image_path(root, source)` in place of the loop, then keeps its existing `Missing Frame desktop alias target` check. Add `python3 tests/frame-desktop-symlink-test.py` to build_files/check.sh after the `recovery-test.py` line.

```python
"""Frame desktop alias resolution stays inside the image root."""
import importlib.util
import os
import tempfile
from pathlib import Path

spec = importlib.util.spec_from_file_location(
    "stage_frame_desktop", Path(__file__).parents[1] / "build_files" / "stage-frame-desktop.py")
stage = importlib.util.module_from_spec(spec)
spec.loader.exec_module(stage)

with tempfile.TemporaryDirectory() as directory:
    base = Path(directory).resolve()
    root = base / "image"
    (root / "usr/lib").mkdir(parents=True)
    (root / "usr/bin").mkdir(parents=True)
    (root / "etc").mkdir(parents=True)
    (base / "host-secret").write_text("outside the image")
    (root / "usr/lib/libreal.so.1").write_bytes(b"\x7fELFreal")
    (root / "etc/shadow").write_text("inside the image")
    os.symlink("libreal.so.1", root / "usr/lib/libalias.so.1")      # relative, stays inside
    os.symlink("/etc/shadow", root / "usr/lib/absolute-alias")      # absolute, re-rooted into the image
    os.symlink("../../../host-secret", root / "usr/bin/escape")     # relative, climbs out of the image
    os.symlink("/usr/lib/libalias.so.1", root / "usr/bin/chain")    # two links deep

    assert stage.resolve_image_path(root, root / "usr/lib/libalias.so.1") == root / "usr/lib/libreal.so.1"
    assert stage.resolve_image_path(root, root / "usr/lib/absolute-alias") == root / "etc/shadow"
    assert stage.resolve_image_path(root, root / "usr/bin/chain") == root / "usr/lib/libreal.so.1"
    assert stage.resolve_image_path(root, root / "etc/shadow") == root / "etc/shadow"
    try:
        stage.resolve_image_path(root, root / "usr/bin/escape")
        raise AssertionError("Symlink outside the image accepted")
    except ValueError as error:
        assert "escapes" in str(error), str(error)

print("Frame desktop alias resolution stays inside the image root.")
```
