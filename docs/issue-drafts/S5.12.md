What: I append the Proton game log to a fixed name in /tmp without checking whether that name is a symlink. A link at that name sends the log text into the link's target file.
Where: packages/quest-runtime/linux/proton-game-launcher.py:422-431 (appid check and log open in main()) (code map: docs/code-map.md#quest-linux-session)
Size: Easy (about 8 changed lines; 1 file; Python; host test)
Serves: none (cleanup)
Done when: open_game_log() opens the log with O_NOFOLLOW, O_APPEND and mode 0600. A symlink at the log path raises OSError with errno ELOOP and leaves its target unchanged. main() calls the helper. The locked test file passes. Whether another local account can create that link depends on the kernel's protected_symlinks setting, which the repo does not set, so the check is about the link itself.

## Qwen brief
Expectation:
- open_game_log('504230', directory) returns a file descriptor for directory/quest-proton-504230.log. Writing b'first\n' and then calling it again to write b'second\n' gives a file with content b'first\nsecond\n'.
- The log file mode is 0o600.
- A symlink at directory/quest-proton-504230.log that points to victim.txt (content 'keep'): open_game_log raises OSError whose errno is ELOOP, and victim.txt still reads 'keep'.
- The default directory is /tmp, and the log file name is unchanged.

def main():
    entry = Path(sys.argv[1])
    args = sys.argv[2:]
    if "--" not in args:
        raise ValueError("Missing Steam runtime command")
    args = args[args.index("--") + 1:]
    if not args:
        raise ValueError("Empty Steam runtime command")
    platforms = sorted(entry.parent.glob("*_platform_*/usr-mtree.txt.gz"))
    if not platforms:
        raise ValueError("Steam ARM64 platform is not installed")
    appid = os.environ.get("SteamAppId", os.environ.get("STEAM_COMPAT_APP_ID", "0"))
    if not appid.isdecimal():
        raise ValueError("Invalid Steam application ID")
    # Steam can pass an epoll descriptor as stdout. Always give Wine ordinary
    # log descriptors; writing a line to epoll fails with EINVAL.
    fd = os.open("/tmp/quest-proton-" + appid + ".log", os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o600)
    os.dup2(fd, 1)
    os.dup2(fd, 2)
    if fd > 2:
        os.close(fd)

## Tests (locked)
Path: packages/quest-runtime/tests/proton-game-log.test.py
```python
import errno
import importlib.util
import os
from pathlib import Path
import tempfile
import unittest

source = Path(__file__).resolve().parents[1] / 'linux/proton-game-launcher.py'
spec = importlib.util.spec_from_file_location('proton', source)
proton = importlib.util.module_from_spec(spec)
spec.loader.exec_module(proton)


class GameLogTest(unittest.TestCase):
    def setUp(self):
        directory = tempfile.TemporaryDirectory()
        self.addCleanup(directory.cleanup)
        self.root = Path(directory.name)
        self.log = self.root / 'quest-proton-504230.log'

    def write(self, fd, text):
        os.write(fd, text)
        os.close(fd)

    def test_log_is_created_private_and_appended(self):
        self.write(proton.open_game_log('504230', self.root), b'first\n')
        self.write(proton.open_game_log('504230', self.root), b'second\n')
        self.assertEqual(self.log.read_bytes(), b'first\nsecond\n')
        self.assertEqual(self.log.stat().st_mode & 0o777, 0o600)

    def test_symlink_at_log_path_is_refused_and_target_untouched(self):
        victim = self.root / 'victim.txt'
        victim.write_text('keep')
        os.symlink(victim, self.log)
        with self.assertRaises(OSError) as raised:
            proton.open_game_log('504230', self.root)
        self.assertEqual(raised.exception.errno, errno.ELOOP)
        self.assertEqual(victim.read_text(), 'keep')


if __name__ == '__main__':
    unittest.main()
```
