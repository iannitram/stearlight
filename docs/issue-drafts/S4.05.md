## Qwen brief
Expectation: Add `fetch_archive(url, archive, sha256, name)` to build_files/quest/stage-chromium.py and call it from `stage()` in place of the inline download and checksum code. It returns None.
1. If `archive` is missing: download `url` with `urllib.request.urlopen(url, timeout=60)` into `archive` plus `.partial` (`archive.with_suffix(archive.suffix + '.partial')`). Check the SHA-256 of the `.partial` file. If it matches, `os.replace` it to `archive`.
2. If the download raises (for example `OSError` mid-stream): delete the `.partial` file, re-raise. `archive` must not exist afterwards.
3. If a fresh download fails the checksum: delete the `.partial` file, raise `ValueError('Chromium package checksum mismatch: ' + name)`. `archive` must not exist afterwards.
4. If `archive` exists and its checksum fails: delete `archive`, raise the same `ValueError`.
5. If `archive` exists and its checksum matches: return without any network call.
Keep the `dpkg-deb` call in `stage()` unchanged. Add `python3 tests/chromium-download-test.py` to build_files/check.sh after the `recovery-test.py` line.

```python
def stage(frame, desktop, output, downloads):
    lock = json.loads((Path(__file__).parent / 'dependencies.json').read_text())['chromium']
    output.mkdir(parents=True, exist_ok=True)
    downloads.mkdir(parents=True, exist_ok=True)
    for name, source in lock.items():
        archive = downloads / (source['sha256'][:16] + '-' + source['url'].rsplit('/', 1)[1])
        if not archive.exists():
            with urllib.request.urlopen(source['url'], timeout=60) as response, archive.open('wb') as target:
                shutil.copyfileobj(response, target)
        with archive.open('rb') as stream:
            if hashlib.file_digest(stream, 'sha256').hexdigest() != source['sha256']:
                raise ValueError('Chromium package checksum mismatch: ' + name)
        subprocess.run(['dpkg-deb', '-x', str(archive), str(output)], check=True)
```

```python
"""Chromium archives reach their final name only after a verified, complete download."""
import hashlib
import importlib.util
import tempfile
from pathlib import Path
import urllib.request

spec = importlib.util.spec_from_file_location(
    "stage_chromium", Path(__file__).parents[1] / "build_files" / "quest" / "stage-chromium.py")
stage_chromium = importlib.util.module_from_spec(spec)
spec.loader.exec_module(stage_chromium)

PAYLOAD = b"chromium package " * 4096
DIGEST = hashlib.sha256(PAYLOAD).hexdigest()
URL = "https://example.invalid/pool/chromium_1.deb"
NAME = "chromium"
ORIGINAL_URLOPEN = urllib.request.urlopen


class FakeResponse:
    def __init__(self, data, fail_after=None):
        self.data = data
        self.fail_after = fail_after
        self.offset = 0

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        return False

    def read(self, size=-1):
        if self.fail_after is not None and self.offset >= self.fail_after:
            raise OSError("connection reset")
        end = len(self.data) if size < 0 else min(len(self.data), self.offset + size)
        if self.fail_after is not None:
            end = min(end, self.fail_after)
        chunk = self.data[self.offset:end]
        self.offset = end
        return chunk


def serve(data=PAYLOAD, fail_after=None):
    urllib.request.urlopen = lambda url, timeout=None: FakeResponse(data, fail_after)


def fetch(archive, digest=DIGEST):
    stage_chromium.fetch_archive(URL, archive, digest, NAME)


def no_network(url, timeout=None):
    raise AssertionError("Network used for a cached archive")


with tempfile.TemporaryDirectory() as directory:
    archive = Path(directory) / "0123456789abcdef-chromium_1.deb"
    try:
        # 1. A complete download lands under the final name, with nothing else left behind.
        serve()
        fetch(archive)
        assert archive.read_bytes() == PAYLOAD
        assert sorted(p.name for p in Path(directory).iterdir()) == [archive.name]
        archive.unlink()

        # 2. An interrupted download leaves no file under the final name, and a rerun succeeds.
        serve(fail_after=1000)
        try:
            fetch(archive)
            raise AssertionError("Interrupted download accepted")
        except OSError:
            pass
        assert not archive.exists(), "Partial data under the final name"
        serve()
        fetch(archive)
        assert archive.read_bytes() == PAYLOAD
        archive.unlink()

        # 3. A checksum mismatch raises and removes the bad download.
        serve(data=b"tampered" * 4096)
        try:
            fetch(archive)
            raise AssertionError("Bad checksum accepted")
        except ValueError as error:
            assert str(error) == "Chromium package checksum mismatch: " + NAME, str(error)
        assert not archive.exists(), "Bad download kept"

        # 4. A corrupt archive left by an earlier run is removed, so the next run downloads it again.
        archive.write_bytes(b"corrupt")
        serve()
        try:
            fetch(archive)
            raise AssertionError("Corrupt cached archive accepted")
        except ValueError:
            pass
        assert not archive.exists(), "Corrupt cached archive kept"
        fetch(archive)
        assert archive.read_bytes() == PAYLOAD
        archive.unlink()

        # 5. A verified cached archive is used without any network request.
        archive.write_bytes(PAYLOAD)
        urllib.request.urlopen = no_network
        fetch(archive)
    finally:
        urllib.request.urlopen = ORIGINAL_URLOPEN

print("Chromium download completion, checksum removal and cache reuse passed.")
```
