What: I stop reading the window list when one log line is longer than the 131072-byte read window. Every later poll reads the same bytes, so the window list stays stale until the workspace restarts.
Where: packages/quest-runtime/linux/frame_menus.py:286-290 (FrameMenus.poll) (code map: docs/code-map.md#quest-linux-session)
Size: Easy (about 8 changed lines; 1 file; Python; host test)
Serves: M3
Done when: a STEARLIGHT_WINDOWS line longer than 131072 bytes is consumed (read whole, or skipped with log_offset moved past it), and the next line updates the window list on the same poll; a partial line with no newline is still left for the next poll; the locked test file passes; the existing packages/quest-runtime/tests/frame-menus.test.py still passes.

## Qwen brief
Expectation:
- workspace.log holds LONG (a STEARLIGHT_WINDOWS line of 140000 'x' characters in its title, then a newline) followed by NEXT (a STEARLIGHT_WINDOWS line whose window has title "Terminal").
- After one poll(): status()['windows'] has exactly one window with title "Terminal", and frame.log_offset equals the file size.
- After appending EMPTY (a STEARLIGHT_WINDOWS line with an empty list) and polling again: status()['windows'] is [].
- A file that ends in a line without its newline: poll() leaves log_offset at 0 and the window list empty; after the newline is appended, the next poll shows "Terminal".

    def poll(self):
        self.poll_performance()
        if not self.manager.running('workspace'):
            self.windows.clear()
            self.script_ready, self.log_offset = False, 0
            return
        now = time.monotonic()
        environment = RUNTIME / 'desktop-shell.env'
        if not environment.exists():
            return
        if not self.script_ready and now >= self.script_retry:
            self.script_retry = now + 10
            try:
                self.script(Path('/opt/quest/desktop-windows.js').read_text(), 'stearlight-windows', keep=True)
                self.script_ready = True
            except (OSError, ValueError, subprocess.SubprocessError):
                pass
        try:
            with (self.state / 'workspace.log').open('rb') as log:
                if log.seek(0, 2) < self.log_offset:
                    self.log_offset = 0
                log.seek(self.log_offset)
                lines = log.read(131072).splitlines(keepends=True)
                for line in lines:
                    if not line.endswith(b'\n'):          # lines 288-289: an overlong line stops here forever
                        break
                    self.log_offset += len(line)
                    marker = line.find(b'STEARLIGHT_WINDOWS ')
                    if marker < 0:
                        continue
                    value = json.loads(line[marker + len(b'STEARLIGHT_WINDOWS '):])
                    current = {}
                    for window in value[:128]:
                        uid = str(window['id'])
                        if uid not in self.window_ids:
                            self.window_ids[uid] = WINDOW_BASE + len(self.window_ids)
                        app_id = str(window.get('app_id', '')).removesuffix('.desktop')
                        current[self.window_ids[uid]] = {'id': uid, 'title': str(window['title'])[:512], 'app_id': app_id}
                        if app_id not in self.history:
                            self.remember(app_id)
                    self.windows = current
        except (OSError, ValueError, KeyError, TypeError):
            pass

import importlib.util
from pathlib import Path
import tempfile
import unittest
from unittest.mock import patch

path = Path(__file__).resolve().parents[1] / 'linux/frame_menus.py'
spec = importlib.util.spec_from_file_location('menus', path)
menus = importlib.util.module_from_spec(spec)
spec.loader.exec_module(menus)

LONG = b'js: STEARLIGHT_WINDOWS [{"id":"big","title":"' + b'x' * 140000 + b'"}]\n'
NEXT = b'js: STEARLIGHT_WINDOWS [{"id":"next","title":"Terminal"}]\n'
EMPTY = b'js: STEARLIGHT_WINDOWS []\n'


class OverlongLogLineTest(unittest.TestCase):
    def setUp(self):
        temp = tempfile.TemporaryDirectory()
        self.addCleanup(temp.cleanup)
        self.root = Path(temp.name)
        self.apps = self.root / 'applications'
        self.apps.mkdir()
        pins = patch.multiple(menus, APPLICATIONS=(self.apps,), RUNTIME=self.root,
                              ACTIVITY_DB=self.root / 'activities.sqlite', ICON_ROOT=self.root / 'icons')
        pins.start()
        self.addCleanup(pins.stop)
        self.manager = type('Manager', (), {'processes': {}, 'instances': {},
                                            'running': lambda self, name: True})()
        self.manager.ensure_display = lambda: {'DISPLAY': ':1', 'WAYLAND_DISPLAY': 'gamescope-0'}

        def atomic(target, text):
            target.write_text(text)
        self.frame = menus.FrameMenus(self.manager, self.root, atomic)
        (self.root / 'desktop-shell.env').write_text('environment')
        self.frame.script_ready = True
        self.log = self.root / 'workspace.log'

    def titles(self):
        return [window['title'] for window in self.frame.status()['windows']]

    def test_overlong_line_is_consumed_and_the_next_line_applies(self):
        self.log.write_bytes(LONG + NEXT)
        self.frame.poll()
        self.assertEqual(self.titles(), ['Terminal'])
        self.assertEqual(self.frame.log_offset, len(LONG + NEXT))

    def test_lines_appended_after_the_overlong_line_are_read(self):
        self.log.write_bytes(LONG + NEXT)
        self.frame.poll()
        with self.log.open('ab') as stream:
            stream.write(EMPTY)
        self.frame.poll()
        self.assertEqual(self.frame.status()['windows'], [])

    def test_partial_line_without_newline_waits_for_the_next_poll(self):
        self.log.write_bytes(NEXT[:-1])
        self.frame.poll()
        self.assertEqual(self.frame.log_offset, 0)
        self.assertEqual(self.frame.status()['windows'], [])
        with self.log.open('ab') as stream:
            stream.write(NEXT[-1:])
        self.frame.poll()
        self.assertEqual(self.titles(), ['Terminal'])


if __name__ == '__main__':
    unittest.main()
