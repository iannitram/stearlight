What: I write the VTOL graphics backup straight to its final name, so a write cut off by a crash or a full disk leaves a truncated file that later launches treat as the original.
Where: packages/quest-runtime/linux/proton-game-launcher.py:100-102 (prepare_vtol_textures); the helper to reuse is backup_original() at 140-158 (code map: docs/code-map.md#quest-linux-session)
Size: Easy (about 4 changed lines; 1 file; Python; host test)
Serves: M3
Done when: the VTOL backup is written through backup_original() (temporary file, fsync, rename), a truncated existing backup is replaced by the verified original, and the locked test file passes. The two other plain writes of the same pattern (proton-game-launcher.py:118-120 and 273-275) are not part of this issue.

## Qwen brief
Expectation:
- Game file bytes equal to the original build, with no backup present: after the call, globalgamemanagers.quest-original holds exactly those bytes and globalgamemanagers holds the adjusted bytes.
- The backup is never written by a plain write to a path ending in .quest-original.
- An existing backup that is truncated (the first 1000 bytes) is replaced by the full original.
- Game file already at the adjusted build: the call returns, the valid backup is kept, and nothing else changes.
- Game file of an unknown build with no backup: ValueError, and no backup file is created.

Code map snippet:
```python
def prepare_vtol_textures(game):
    # Unity 2020's textureQuality setting omits the two largest mips at load time.
    # This changes graphics configuration, not game code or stereo geometry.
    asset = game / "VTOLVR_Data/globalgamemanagers"
    original = "141f912ee5e463e962fe81e96009d93b41d3bbead220e4249b814f8644253709"
    previous = "a8eef375dff3597024700f89a56cec91ed916c31e2cdfbcf4a12db711083e3e3"
    adjusted = "437b512ae393213790f787961ae2d827e40da6934ef891aa6798f37534ef41a8"
    data = asset.read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    if digest == adjusted:
        return
    if digest not in (original, previous):
        raise ValueError("VTOL graphics profile does not match this game build")
    backup = asset.with_name(asset.name + ".quest-original")
    if digest == original and not backup.exists():      # lines 101-102: change these two lines
        backup.write_bytes(data)
    updated = data[:259448] + b"\x02\x00\x00\x00" + data[259452:]
    if hashlib.sha256(updated).hexdigest() != adjusted:
        raise ValueError("VTOL graphics profile verification failed")
    temporary = asset.with_name(asset.name + ".quest-next")
    temporary.write_bytes(updated)
    temporary.replace(asset)


def backup_original(path, data, valid):
    # A killed/full-disk write must never become the next launch's original.
    try:
        if valid(path.read_bytes()):
            return
    except FileNotFoundError:
        pass
    if not valid(data):
        raise ValueError("Cannot recover an invalid game backup")
    descriptor, name = tempfile.mkstemp(prefix=path.name + ".", dir=path.parent)
    temporary = Path(name)
    try:
        with os.fdopen(descriptor, "wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        temporary.replace(path)
    finally:
        temporary.unlink(missing_ok=True)
```
The intended change is `backup_original(backup, data, lambda value: hashlib.sha256(value).hexdigest() == original)` in place of lines 101-102.

```python
import hashlib
import importlib.util
from pathlib import Path
import tempfile
import unittest
from unittest.mock import patch

source = Path(__file__).resolve().parents[1] / 'linux/proton-game-launcher.py'
spec = importlib.util.spec_from_file_location('proton', source)
proton = importlib.util.module_from_spec(spec)
spec.loader.exec_module(proton)

ORIGINAL_DIGEST = '141f912ee5e463e962fe81e96009d93b41d3bbead220e4249b814f8644253709'
ADJUSTED_DIGEST = '437b512ae393213790f787961ae2d827e40da6934ef891aa6798f37534ef41a8'
ORIGINAL = bytes(range(256)) * 1100
ADJUSTED = ORIGINAL[:259448] + b'\x02\x00\x00\x00' + ORIGINAL[259452:]
KNOWN = {ORIGINAL: ORIGINAL_DIGEST, ADJUSTED: ADJUSTED_DIGEST}
REAL_SHA256 = hashlib.sha256
REAL_WRITE = Path.write_bytes


class Digest:
    def __init__(self, value):
        self.value = value

    def hexdigest(self):
        return self.value


def fake_sha256(data=b''):
    if bytes(data) in KNOWN:
        return Digest(KNOWN[bytes(data)])
    return REAL_SHA256(data)


def no_in_place_backup(self, data, *args, **kwargs):
    if self.name.endswith('.quest-original'):
        raise OSError('backup written in place')
    return REAL_WRITE(self, data, *args, **kwargs)


class VtolBackupTest(unittest.TestCase):
    def setUp(self):
        directory = tempfile.TemporaryDirectory()
        self.addCleanup(directory.cleanup)
        self.game = Path(directory.name)
        self.asset = self.game / 'VTOLVR_Data/globalgamemanagers'
        self.asset.parent.mkdir(parents=True)
        self.backup = self.asset.with_name(self.asset.name + '.quest-original')
        patcher = patch.object(proton.hashlib, 'sha256', side_effect=fake_sha256)
        patcher.start()
        self.addCleanup(patcher.stop)

    def run_with(self, data):
        self.asset.write_bytes(data)
        proton.prepare_vtol_textures(self.game)

    def test_first_run_saves_verified_original_and_adjusts_game_file(self):
        self.run_with(ORIGINAL)
        self.assertEqual(self.backup.read_bytes(), ORIGINAL)
        self.assertEqual(self.asset.read_bytes(), ADJUSTED)

    def test_backup_is_never_written_in_place(self):
        with patch.object(Path, 'write_bytes', no_in_place_backup):
            self.run_with(ORIGINAL)
        self.assertEqual(self.backup.read_bytes(), ORIGINAL)

    def test_truncated_backup_is_replaced_by_the_verified_original(self):
        self.backup.write_bytes(ORIGINAL[:1000])
        self.run_with(ORIGINAL)
        self.assertEqual(self.backup.read_bytes(), ORIGINAL)

    def test_valid_backup_is_kept_when_game_file_is_already_adjusted(self):
        self.backup.write_bytes(ORIGINAL)
        self.run_with(ADJUSTED)
        self.assertEqual(self.backup.read_bytes(), ORIGINAL)
        self.assertEqual(self.asset.read_bytes(), ADJUSTED)

    def test_unknown_game_build_is_refused_and_creates_no_backup(self):
        with self.assertRaises(ValueError):
            self.run_with(b'other build')
        self.assertFalse(self.backup.exists())


if __name__ == '__main__':
    unittest.main()
```
