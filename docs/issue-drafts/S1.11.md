What: I return a busy status (exit 75) when another SteamOS update holds the lock, and I leave progress.json alone. Today a second run writes "SteamOS update failed" while the first run is still downloading.
Where: packages/quest-runtime/linux/steam-os-update.py:333-336 (lock) and 354-364 (top-level handler) (code map: docs/code-map.md#steam-update)
Size: Easy (about 8 changed lines; 1 file; Python; host test)

Done when: a second main() that finds the lock held returns 75 without raising, prints one busy line to stderr, and does not touch progress.json; the locked test file passes; the existing packages/quest-runtime/tests/steam-os-update.test.py still passes.

## Qwen brief
Expectation:
- The lock file is held by another open file description with LOCK_EX | LOCK_NB. The test does this.
- main(['check']) returns 75 and does not raise.
- candidate() is not called, and progress.json keeps its exact bytes.
- Stderr gets one line that starts with "Another SteamOS update is running".
- When the lock is free, main(['check']) acquires it and reaches candidate(). The test makes candidate() raise RuntimeError('reached'), and main() propagates it.

## Tests (locked)
Path: packages/quest-runtime/tests/steam-os-update-lock.test.py
```python
import fcntl
import importlib.util
from pathlib import Path
import tempfile
import unittest
from unittest.mock import patch

source = Path(__file__).resolve().parents[1] / 'linux/steam-os-update.py'
spec = importlib.util.spec_from_file_location('osupdate', source)
update = importlib.util.module_from_spec(spec)
spec.loader.exec_module(update)

PROGRESS = b'{"title": "Downloading", "detail": "", "fraction": 0.4, "error": false}\n'


class BusyLockTest(unittest.TestCase):
    def setUp(self):
        directory = tempfile.TemporaryDirectory()
        self.addCleanup(directory.cleanup)
        self.cache = Path(directory.name)
        (self.cache / 'progress.json').write_bytes(PROGRESS)
        self.holder = (self.cache / 'lock').open('a')
        self.addCleanup(self.holder.close)
        cache = patch.object(update, 'CACHE', self.cache)
        cache.start()
        self.addCleanup(cache.stop)

    def hold(self):
        fcntl.flock(self.holder, fcntl.LOCK_EX | fcntl.LOCK_NB)

    def test_busy_lock_returns_busy_status(self):
        self.hold()
        with patch.object(update, 'candidate') as candidate:
            self.assertEqual(update.main(['check']), 75)
        candidate.assert_not_called()

    def test_busy_lock_leaves_progress_file_untouched(self):
        self.hold()
        update.main(['check'])
        self.assertEqual((self.cache / 'progress.json').read_bytes(), PROGRESS)

    def test_free_lock_is_acquired_and_work_starts(self):
        self.hold()
        fcntl.flock(self.holder, fcntl.LOCK_UN)
        with patch.object(update, 'candidate', side_effect=RuntimeError('reached')):
            with self.assertRaisesRegex(RuntimeError, 'reached'):
                update.main(['check'])


if __name__ == '__main__':
    unittest.main()
```
