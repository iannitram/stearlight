"""Exercise APK overlay extraction with and without an installed OS update."""
import io
import os
from pathlib import Path
import shutil
import subprocess
import tarfile
import tempfile
import unittest

OS_STUB = '''package android.system;
public class Os {
 public static void chmod(String p,int m) {}
 public static void symlink(String a,String b) throws Exception {throw new Exception("unexpected symlink");}
 public static void link(String a,String b) throws ErrnoException {
  ErrnoException error=new ErrnoException();error.errno=Integer.getInteger("stub.linkErrno",13);throw error;
 }
 public static Stat stat(String p) {return new Stat();}
 public static class Stat {public int st_mode=0644;}
}'''
ERRNO_STUB = 'package android.system; public class ErrnoException extends Exception {public int errno;}'
CONSTANTS_STUB = 'package android.system; public class OsConstants {public static int EACCES=13,EPERM=1;}'
MAIN = '''package net.stearlight.quest;
import java.io.*;
public class ArchiveTest {
 public static void main(String[] args) throws Exception {
  RootfsArchive.extract(new FileInputStream(args[0]),new File(args[1]),Boolean.parseBoolean(args[2]));
 }
}'''


class RootfsArchiveTest(unittest.TestCase):
    RUNTIME = 'opt/steamvr/resources/' + 'long-path-' * 14 + '/panel.js'

    def setUp(self):
        java_home = os.environ.get('JAVA_HOME')
        self.javac = str(Path(java_home) / 'bin/javac') if java_home else shutil.which('javac')
        self.java = str(Path(java_home) / 'bin/java') if java_home else shutil.which('java')
        self.assertTrue(self.javac and self.java, 'JDK 17 is required')
        directory = tempfile.TemporaryDirectory()
        self.addCleanup(directory.cleanup)
        self.root = Path(directory.name)
        self.compile()

    def compile(self):
        package = self.root / 'net/stearlight/quest'
        android = self.root / 'android/system'
        package.mkdir(parents=True)
        android.mkdir(parents=True)
        source = Path(__file__).resolve().parents[1] / 'app/src/main/java/net/stearlight/quest/RootfsArchive.java'
        shutil.copy2(source, package / source.name)
        (android / 'Os.java').write_text(OS_STUB)
        (android / 'ErrnoException.java').write_text(ERRNO_STUB)
        (android / 'OsConstants.java').write_text(CONSTANTS_STUB)
        (package / 'ArchiveTest.java').write_text(MAIN)
        subprocess.run([self.javac, '-d', str(self.root), *map(str, self.root.rglob('*.java'))], check=True)

    def write_archive(self, path, entries):
        with tarfile.open(path, 'w:gz', format=tarfile.PAX_FORMAT) as output:
            for name, data, linkname in entries:
                entry = tarfile.TarInfo(name)
                if linkname:
                    entry.type = tarfile.LNKTYPE
                    entry.linkname = linkname
                else:
                    entry.size = len(data)
                output.addfile(entry, io.BytesIO(data or b''))
        return path

    def standard_entries(self):
        return [
            (self.RUNTIME, b'bundled runtime' * 37, None),
            ('opt/quest/start-session.sh', b'new launcher', None),
            ('opt/quest/start-link.sh', None, 'opt/quest/start-session.sh'),
        ]

    def extract(self, archive, destination, preserve, link_errno=13):
        return subprocess.run([self.java, f'-Dstub.linkErrno={link_errno}', '-cp', str(self.root),
                               'net.stearlight.quest.ArchiveTest', str(archive), str(destination),
                               str(preserve).lower()], capture_output=True, text=True)

    def test_overlay_preserves_updated_runtime_and_extracts_following_entries(self):
        archive = self.write_archive(self.root / 'overlay.tar.gz', self.standard_entries())
        for preserve in (False, True):
            with self.subTest(preserve=preserve):
                destination = self.root / str(preserve)
                runtime = destination / self.RUNTIME
                runtime.parent.mkdir(parents=True)
                runtime.write_bytes(b'installed OS runtime')
                result = self.extract(archive, destination, preserve)
                self.assertEqual(result.returncode, 0, result.stderr)
                expected = b'installed OS runtime' if preserve else b'bundled runtime' * 37
                self.assertEqual(runtime.read_bytes(), expected)
                self.assertEqual((destination / 'opt/quest/start-session.sh').read_bytes(), b'new launcher')
                self.assertEqual((destination / 'opt/quest/start-link.sh').read_bytes(), b'new launcher')

    def test_hardlink_fallback_only_for_permission_errors(self):
        archive = self.write_archive(self.root / 'links.tar.gz', self.standard_entries())
        for errno in (13, 1):  # EACCES, EPERM
            with self.subTest(errno=errno):
                destination = self.root / f'errno-{errno}'
                result = self.extract(archive, destination, False, errno)
                self.assertEqual(result.returncode, 0, result.stderr)
                self.assertEqual((destination / 'opt/quest/start-link.sh').read_bytes(), b'new launcher')
        result = self.extract(archive, self.root / 'exdev', False, 18)  # EXDEV
        self.assertNotEqual(result.returncode, 0)
        self.assertIn('Cannot create rootfs hardlink opt/quest/start-link.sh', result.stderr)


if __name__ == '__main__':
    unittest.main()
