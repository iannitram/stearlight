What: I write the temporary file for an atomic replace through whatever already sits at its name. A symlink left at that name sends the write to any file the manager can write, and the manager runs as root.
Where: packages/quest-runtime/linux/desktop-manager.py:39-46 (atomic()) (code map: docs/code-map.md#quest-linux-session)
Size: Easy (about 5 changed lines; 1 file; Python; host test)
Serves: none (cleanup)
Done when: atomic() removes whatever is at the temporary name, then creates it exclusively with O_EXCL and O_NOFOLLOW. A symlink at the temporary name is never followed. The final file has the requested mode, and the locked test file passes.

## Qwen brief
Expectation:
- atomic(target, 'secret', 0o600) with no leftover: target reads 'secret', its mode is 0o600, and no file named settings.next remains.
- A regular file at settings.next with mode 0o644 and content 'old': after the call, target reads 'secret', its mode is 0o600, and settings.next does not exist.
- A symlink at settings.next that points to victim.txt (content 'keep'): after the call, victim.txt still reads 'keep', target reads 'secret', and settings.next does not exist.
- The temporary name stays path.with_suffix('.next'), and the replace stays os.replace semantics.

Code map snippet:
```python
def atomic(path, value, mode=0o600):
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix('.next')
    descriptor = os.open(temporary, os.O_CREAT | os.O_TRUNC | os.O_WRONLY, mode)
    with os.fdopen(descriptor, 'w') as stream:
        stream.write(value)
    os.chmod(temporary, mode)
    temporary.replace(path)
```
Imports already present: os, pathlib.Path. The new body keeps the signature `atomic(path, value, mode=0o600)`.

## Tests (locked)

```python
import importlib.util
import os
from pathlib import Path
import tempfile
import unittest

source = Path(__file__).resolve().parents[1] / 'linux/desktop-manager.py'
spec = importlib.util.spec_from_file_location('desktop', source)
desktop = importlib.util.module_from_spec(spec)
spec.loader.exec_module(desktop)


class AtomicWriteTest(unittest.TestCase):
    def setUp(self):
        directory = tempfile.TemporaryDirectory()
        self.addCleanup(directory.cleanup)
        self.root = Path(directory.name)
        self.target = self.root / 'settings.json'
        self.temporary = self.root / 'settings.next'

    def test_plain_write_creates_private_file_and_no_leftover(self):
        desktop.atomic(self.target, 'secret', 0o600)
        self.assertEqual(self.target.read_text(), 'secret')
        self.assertEqual(self.target.stat().st_mode & 0o777, 0o600)
        self.assertFalse(self.temporary.exists())

    def test_leftover_regular_file_is_replaced_with_requested_mode(self):
        self.temporary.write_text('old')
        os.chmod(self.temporary, 0o644)
        desktop.atomic(self.target, 'secret', 0o600)
        self.assertEqual(self.target.read_text(), 'secret')
        self.assertEqual(self.target.stat().st_mode & 0o777, 0o600)
        self.assertFalse(self.temporary.exists())

    def test_symlink_at_temporary_name_is_not_followed(self):
        victim = self.root / 'victim.txt'
        victim.write_text('keep')
        os.symlink(victim, self.temporary)
        desktop.atomic(self.target, 'secret', 0o600)
        self.assertEqual(victim.read_text(), 'keep')
        self.assertEqual(self.target.read_text(), 'secret')
        self.assertFalse(self.temporary.exists())


if __name__ == '__main__':
    unittest.main()
```
