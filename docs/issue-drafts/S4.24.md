What: I want the container-input check in the layout test to skip `--flag=` options on COPY lines and to check sources written in the JSON-array COPY form. Today `COPY --chmod=0755 a /usr/bin/` makes the check look for a file named `--chmod=0755`, so the test fails on the current tree, while `COPY ["a", "/usr/bin/"]` yields no sources and is never checked.
Where: tests/layout-test.sh:19-24 (the Containerfile COPY loop); Containerfile:138-139 (the `--chmod=0755` COPY lines) (code map: docs/code-map.md#tests)
Size: Easy (about 12 changed lines; Python inside a shell script; Linux build host)
Serves: none (cleanup)
Done when: `sh tests/layout-test.sh` passes on the current tree, and `python3 -I tests/layout-container-copy-test.py` passes all six tests. Four of those tests fail on the current code, so the fix is shown to change the result.

## Qwen brief
Expectation: The COPY loop in tests/layout-test.sh must check exactly the sources of each non-`--from=` COPY line, in both forms.
- `COPY --chmod=0755 present.sh /usr/bin/` with present.sh in the tree: exit 0, stdout ends with `Stearlight OS layout and container inputs passed.`
- `COPY --chmod=0755 absent.sh /usr/bin/`: exit 1, stderr contains `Missing container input: absent.sh`.
- `COPY ["absent.json", "/etc/x.json"]`: exit 1, stderr contains `Missing container input: absent.json`.
- `COPY ["present.json", "/etc/x.json"]` with present.json in the tree: exit 0.
- `COPY --from=builder /out/x /x`: skipped, exit 0 even when /out/x does not exist.
- Tokens starting with `--` are options, never sources. The last element (the destination) is never a source.
- The check must still run on the real tree and pass there once the real Containerfile is correct.
Code map snippet (tests/layout-test.sh, lines 19-24; the heredoc reads the repo root from argv[1] and must keep that signature):
```python
python3 - "$repo" <<'PY'
import json, pathlib, shlex, sys
root = pathlib.Path(sys.argv[1])
# ... disk.json check above ...
for line in (root / 'Containerfile').read_text().splitlines():
    if not line.startswith('COPY ') or '--from=' in line:
        continue
    for source in shlex.split(line)[1:-1]:
        assert (root / source).exists(), f'Missing container input: {source}'
PY
```

## Tests (locked)
Path: tests/layout-container-copy-test.py
```python
"""Locked tests for the COPY source check in tests/layout-test.sh.

Each case builds a sandbox from the repo's own inputs, writes a Containerfile
with one COPY form, and runs the real layout-test.sh inside the sandbox.
"""
import shutil
import subprocess
import tempfile
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
# The same files that layout-test.sh requires on line 4.
INPUTS = [
    'Containerfile', 'Justfile', 'build_files/build-image.sh', 'build_files/check.sh',
    'build_files/fetch-recovery.py', 'packages/sources.lock.json',
    'packages/stearlight-runtime/dashboard_ready.c', 'packages/windows-runtime/build-runtime.sh',
    'disk_config/disk.json', 'post_process/start-vm.ps1', 'post_process/start-vm.cmd',
    'post_process/package-windows.ps1', 'system_files/etc/local.d/arm64-vm.start',
    'tests/launcher-test.ps1',
]


def run_layout(containerfile, present=()):
    tmp = Path(tempfile.mkdtemp())
    try:
        for rel in INPUTS:
            (tmp / rel).parent.mkdir(parents=True, exist_ok=True)
            shutil.copy(ROOT / rel, tmp / rel)
        (tmp / 'Containerfile').write_text(containerfile)
        for rel in present:
            (tmp / rel).parent.mkdir(parents=True, exist_ok=True)
            (tmp / rel).write_text('x\n')
        (tmp / 'tests').mkdir(exist_ok=True)
        shutil.copy(ROOT / 'tests/layout-test.sh', tmp / 'tests/layout-test.sh')
        return subprocess.run(['sh', str(tmp / 'tests/layout-test.sh')],
                              capture_output=True, text=True)
    finally:
        shutil.rmtree(tmp, ignore_errors=True)


class CopySources(unittest.TestCase):
    def test_flag_is_not_a_source(self):
        r = run_layout('COPY --chmod=0755 present.sh /usr/bin/\n', ['present.sh'])
        self.assertEqual(r.returncode, 0, r.stderr)
        self.assertIn('passed', r.stdout)

    def test_missing_source_after_flag_fails(self):
        r = run_layout('COPY --chmod=0755 absent.sh /usr/bin/\n')
        self.assertNotEqual(r.returncode, 0)
        self.assertIn('Missing container input: absent.sh', r.stderr)

    def test_json_form_missing_source_fails(self):
        r = run_layout('COPY ["absent.json", "/etc/x.json"]\n')
        self.assertNotEqual(r.returncode, 0)
        self.assertIn('Missing container input: absent.json', r.stderr)

    def test_json_form_present_source_passes(self):
        r = run_layout('COPY ["present.json", "/etc/x.json"]\n', ['present.json'])
        self.assertEqual(r.returncode, 0, r.stderr)

    def test_from_stage_copy_is_skipped(self):
        r = run_layout('COPY --from=builder /out/x /x\n')
        self.assertEqual(r.returncode, 0, r.stderr)

    def test_real_tree_passes(self):
        r = subprocess.run(['sh', str(ROOT / 'tests/layout-test.sh')],
                           capture_output=True, text=True)
        self.assertEqual(r.returncode, 0, r.stderr)


if __name__ == '__main__':
    unittest.main()
```
