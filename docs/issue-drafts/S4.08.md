## Qwen brief
Expectation: Add `html_attribute(value)` to generate_compatibility.py. It returns `html.escape(str(value), quote=True)`, so `"` becomes `&quot;`, `&` becomes `&amp;`, `<` and `>` are escaped, and `'` becomes `&#x27;`.
In `main()`, keep the table name cell as it is, built from `escape_cell(raw)`, where `raw` is the value `steam_name(...)` returned. Change only the cover tag:
- `src` uses `html_attribute(cover_url)`.
- `alt` is `html_attribute(raw)` with newlines replaced by spaces, then `|` replaced by `&#124;`, then ` cover` appended.
Example: raw title `Say "Hi" | Co` with artwork `https://cdn.example.invalid/a"b.jpg` gives
`<img src="https://cdn.example.invalid/a&quot;b.jpg" alt="Say &quot;Hi&quot; &#124; Co cover" width="62">`.
Each table row still has exactly four `|` characters. Add `import html` at the top. Add `python3 tests/generate-compatibility-test.py` to build_files/check.sh after the `recovery-test.py` line.

```python
"""Compatibility table cover images keep their attributes intact when a value contains quotes or pipes."""
import importlib.util
import json
import tempfile
from pathlib import Path

spec = importlib.util.spec_from_file_location(
    "generate_compatibility", Path(__file__).parents[1] / "generate_compatibility.py")
gen = importlib.util.module_from_spec(spec)
spec.loader.exec_module(gen)

assert gen.html_attribute('a"b&c<d>') == 'a&quot;b&amp;c&lt;d&gt;'
assert gen.html_attribute("plain text") == "plain text"

with tempfile.TemporaryDirectory() as directory:
    root = Path(directory)
    games = [
        {"appid": 1, "status": "verified", "name": 'Say "Hi" | Co', "notes": "ok",
         "artwork_url": 'https://cdn.example.invalid/a"b.jpg'},
        {"appid": 2, "status": "playable", "name": "Plain Game", "notes": "fine"},
    ]
    source = root / "games.json"
    source.write_text(json.dumps(games), encoding="utf-8")
    gen.SOURCE = source
    gen.OUTPUT = root / "STANDALONE_COMPATIBILITY.md"
    gen.steam_name = lambda appid, fallback: fallback  # no network in the test
    gen.main()
    text = gen.OUTPUT.read_text(encoding="utf-8")
    rows = [line for line in text.splitlines() if line.startswith("| <img")]
    assert len(rows) == 2, rows

    first = next(row for row in rows if "Say" in row)
    assert 'src="https://cdn.example.invalid/a&quot;b.jpg"' in first, first
    assert 'alt="Say &quot;Hi&quot; &#124; Co cover"' in first, first
    assert first.count("|") == 4, first  # the three column separators and the row's outer pipe

    second = next(row for row in rows if "Plain Game" in row)
    assert 'alt="Plain Game cover"' in second, second
    assert second.count("|") == 4, second

print("Compatibility cover attributes are escaped and table rows keep their column count.")
```
